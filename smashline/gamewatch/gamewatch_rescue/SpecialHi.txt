unsafe extern "C" fn game_specialhi(agent: &mut L2CAgentBase) {
    if macros::is_excute(agent) {
        VisibilityModule::set_int64(agent.module_accessor, hash40("up"), hash40("off"));
        VisibilityModule::set_int64(agent.module_accessor, hash40("down"), hash40("off"));
    }
    frame(agent.lua_state_agent, 1.0);
    if macros::is_excute(agent) {
        VisibilityModule::set_int64(agent.module_accessor, hash40("down"), hash40("on"));
    }
    frame(agent.lua_state_agent, 8.0);
    if macros::is_excute(agent) {
        VisibilityModule::set_int64(agent.module_accessor, hash40("up"), hash40("on"));
        VisibilityModule::set_int64(agent.module_accessor, hash40("down"), hash40("off"));
    }
}
