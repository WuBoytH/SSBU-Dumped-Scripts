unsafe extern "C" fn expression_lose(agent: &mut L2CAgentBase) {
    if macros::is_excute(agent) {
        VisibilityModule::set_mesh_visibility(agent.module_accessor, Hash40::new("head"), *MESH_VISIBILITY_NONE, false);
        VisibilityModule::set_mesh_visibility(agent.module_accessor, Hash40::new("lhand"), *MESH_VISIBILITY_NONE, false);
        VisibilityModule::set_mesh_visibility(agent.module_accessor, Hash40::new("rhand"), *MESH_VISIBILITY_NONE, false);
        VisibilityModule::set_mesh_visibility(agent.module_accessor, Hash40::new("hand"), *MESH_VISIBILITY_NONE, false);
    }
}
