unsafe extern "C" fn effect_attackairn_set(agent: &mut L2CAgentBase) {
    if !WorkModule::is_flag(agent.module_accessor, *FIGHTER_INSTANCE_WORK_ID_FLAG_METAL) {
        if !WorkModule::is_flag(agent.module_accessor, *FIGHTER_INSTANCE_WORK_ID_FLAG_GOLD) {
            if !WorkModule::is_flag(agent.module_accessor, *FIGHTER_INSTANCE_WORK_ID_FLAG_SPYCLOAK) {
                if !macros::FT_IS_SAME_FIGHTER_CATEGORY(agent, *smash::lib::lua_const::FighterCategory::METAMON) {
                    if !macros::FT_IS_SAME_FIGHTER_CATEGORY(agent, *smash::lib::lua_const::FighterCategory::LIGHT) {
                        if !macros::FT_IS_SAME_FIGHTER_CATEGORY(agent, *smash::lib::lua_const::FighterCategory::DARK) {
                            if macros::is_excute(agent) {
                                macros::EFFECT_FOLLOW(agent, Hash40::new("kamui_transform_wind_s"), Hash40::new("havel"), 0, 0, 1, 0, 0, 180, 0.45, true);
                            }
                        }
                    }
                }
            }
        }
    }
}
