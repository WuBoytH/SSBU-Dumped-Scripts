unsafe extern "C" fn effect_specialswall_set(agent: &mut L2CAgentBase) {
    if !WorkModule::is_flag(agent.module_accessor, *FIGHTER_INSTANCE_WORK_ID_FLAG_METAL) {
        if !WorkModule::is_flag(agent.module_accessor, *FIGHTER_INSTANCE_WORK_ID_FLAG_GOLD) {
            if !WorkModule::is_flag(agent.module_accessor, *FIGHTER_INSTANCE_WORK_ID_FLAG_SPYCLOAK) {
                if !macros::FT_IS_SAME_FIGHTER_CATEGORY(agent, *FIGHTER_CATEGORY_METAMON) {
                    if !macros::FT_IS_SAME_FIGHTER_CATEGORY(agent, *FIGHTER_CATEGORY_LIGHT) {
                        if !macros::FT_IS_SAME_FIGHTER_CATEGORY(agent, *FIGHTER_CATEGORY_DARK) {
                            if macros::is_excute(agent) {
                                macros::EFFECT_FOLLOW(agent, Hash40::new_raw(0x1bdac6e17a), Hash40::new_raw(0x05eb263e0d), -3.3, 0, 0, 0, 90, 0, 0.95, true);
                                macros::LAST_EFFECT_SET_RATE(agent, 0.001);
                            }
                        }
                    }
                }
            }
        }
    }
}
