unsafe extern "C" fn game_finalstartl(agent: &mut L2CAgentBase) {
    if macros::is_excute(agent) {
        MotionModule::set_rate(agent.module_accessor, 1.5);
    }
    frame(agent.lua_state_agent, 1.0);
    if macros::is_excute(agent) {
        macros::FT_START_CUTIN(agent);
        ArticleModule::change_status(agent.module_accessor, *FIGHTER_PIKMIN_GENERATE_ARTICLE_DOLFIN, *WEAPON_PIKMIN_DOLFIN_STATUS_KIND_START, ArticleOperationTarget(0));
        macros::SLOW_OPPONENT(agent, 5.0, 40.0);
        macros::CHECK_VALID_FINAL_START_CAMERA(agent, 0, 7, 20, 0, 0, 0);
    }
    if !WorkModule::is_flag(agent.module_accessor, *FIGHTER_INSTANCE_WORK_ID_FLAG_DISABLE_FINAL_START_CAMERA) {
        if macros::is_excute(agent) {
            macros::FT_SET_FINAL_FEAR_FACE(agent, 60);
            macros::REQ_FINAL_START_CAMERA(agent, Hash40::new("d04finalstartr.nuanmb"), false, false) /* TODO: macro expects 2 args */;
        }
    } else {
        if macros::is_excute(agent) {
            let scale = PostureModule::scale(agent.module_accessor);
            macros::CAM_ZOOM_IN(agent, 5, 0, scale * 2.2, 0, 0);
        }
    }
    frame(agent.lua_state_agent, 41.0);
    if macros::is_excute(agent) {
        MotionModule::set_rate(agent.module_accessor, 1.0);
        macros::CAM_ZOOM_OUT(agent);
    }
}
