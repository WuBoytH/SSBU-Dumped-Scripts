unsafe extern "C" fn game_finalair(agent: &mut L2CAgentBase) {
    if macros::is_excute(agent) {
        ArticleModule::generate_article(agent.module_accessor, *FIGHTER_NESS_GENERATE_ARTICLE_PAULA, false, -1);
        ArticleModule::generate_article(agent.module_accessor, *FIGHTER_NESS_GENERATE_ARTICLE_POO, false, -1);

        article!(agent, *MA_MSC_CMD_ARTICLE_SET_OFFSET_ARTICLE, *FIGHTER_NESS_GENERATE_ARTICLE_POO, 10, 0, -0.01);

        macros::SLOW_OPPONENT(agent, 5.0, 30.0);
    }
    if lua_bind::BattleObjectWorld::scale_z(singletons::BattleObjectWorld()) <= 1.0 {
        article!(agent, *MA_MSC_CMD_ARTICLE_SET_OFFSET_ARTICLE, *FIGHTER_NESS_GENERATE_ARTICLE_PAULA, -10, 0, -10);
        article!(agent, *MA_MSC_CMD_ARTICLE_SET_OFFSET_ARTICLE, *FIGHTER_NESS_GENERATE_ARTICLE_POO, 10, 0, -10);
    } else {
    }
    if !WorkModule::is_flag(agent.module_accessor, *FIGHTER_INSTANCE_WORK_ID_FLAG_DISABLE_FINAL_START_CAMERA) {
        frame(agent.lua_state_agent, 10.0);
        if macros::is_excute(agent) {
            macros::FT_SET_FINAL_FEAR_FACE(agent, 60);
            macros::REQ_FINAL_START_CAMERA(agent, Hash40::new("d04final.nuanmb"), false);
        }
    } else {
        if macros::is_excute(agent) {
            let scale = PostureModule::scale(agent.module_accessor);
            macros::CAM_ZOOM_IN(agent, 5, 0, scale * 2.2, 0, 0);
        }
    }
    macros::FT_START_CUTIN(agent);
    frame(agent.lua_state_agent, 25.0);
    if macros::is_excute(agent) {
        macros::CAM_ZOOM_OUT(agent);
        camera!(agent, *MA_MSC_CMD_CAMERA_CAM_RECT, 50, -50, 50, -10);
    }
    frame(agent.lua_state_agent, 51.0);
    if macros::is_excute(agent) {
        WorkModule::on_flag(agent.module_accessor, *FIGHTER_NESS_INSTANCE_WORK_ID_FLAG_FINAL_START);
    }
    frame(agent.lua_state_agent, 350.0);
    if macros::is_excute(agent) {
        ArticleModule::remove_exist(agent.module_accessor, *FIGHTER_NESS_GENERATE_ARTICLE_PAULA, ArticleOperationTarget(0));
        ArticleModule::remove_exist(agent.module_accessor, *FIGHTER_NESS_GENERATE_ARTICLE_POO, ArticleOperationTarget(0));
    }
    frame(agent.lua_state_agent, 357.0);
    if macros::is_excute(agent) {
        cancel!(agent, *MA_MSC_CMD_CANCEL_ENABLE_CANCEL);
    }
}
