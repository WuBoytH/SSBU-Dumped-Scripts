unsafe extern "C" fn game_hit(agent: &mut L2CAgentBase) {
    if WorkModule::is_flag(agent.module_accessor, *WEAPON_ELIGHT_BUNSHIN_INSTANCE_WORK_ID_FLAG_FINISH) {
        if macros::is_excute(agent) {
            ControlModule::set_rumble(agent.module_accessor, Hash40::new_raw(0x14b87addf2), 0, false, *BATTLE_OBJECT_ID_INVALID as u32);
        }
    } else {
        if macros::is_excute(agent) {
            ControlModule::set_rumble(agent.module_accessor, Hash40::new_raw(0x0db75641a5), 5, false, *BATTLE_OBJECT_ID_INVALID as u32);
        }
    }
}
