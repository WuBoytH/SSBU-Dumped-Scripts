unsafe extern "C" fn game_finalstart(agent: &mut L2CAgentBase) {
    if macros::is_excute(agent) {
        macros::CHECK_VALID_FINAL_START_CAMERA(agent, 0, 7, 20, 0, 0, 0);
        macros::SLOW_OPPONENT(agent, 4.0, 50.0);
    }
    if !WorkModule::is_flag(agent.module_accessor, *FIGHTER_INSTANCE_WORK_ID_FLAG_DISABLE_FINAL_START_CAMERA) {
        frame(agent.lua_state_agent, 10.0);
        if macros::is_excute(agent) {
            macros::FT_SET_FINAL_FEAR_FACE(agent, 60);
        }
        if get_value_float(agent.lua_state_agent, *SO_VAR_FLOAT_LR) < 0.0 {
            if macros::is_excute(agent) {
                macros::REQ_FINAL_START_CAMERA(agent, Hash40::new_raw(0x1654ea70d3), false, false) /* TODO: macro expects 2 args */;
            }
        } else {
            if macros::is_excute(agent) {
                macros::REQ_FINAL_START_CAMERA(agent, Hash40::new_raw(0x14c2bd844e), false, false) /* TODO: macro expects 2 args */;
            }
        }
    } else {
        if macros::is_excute(agent) {
            let scale = PostureModule::scale(agent.module_accessor);
            macros::CAM_ZOOM_IN(agent, 5, 0, scale * 1.8, 0, 0);
        }
    }
    if macros::is_excute(agent) {
        macros::FT_START_CUTIN(agent);
    }
    frame(agent.lua_state_agent, 25.0);
    if macros::is_excute(agent) {
        macros::CAM_ZOOM_OUT(agent);
    }
}
