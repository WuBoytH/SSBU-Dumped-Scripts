    }

            if !WorkModule::is_flag(boma, *FIGHTER_INSTANCE_WORK_ID_FLAG_SPYCLOAK) {

                                EFFECT_FOLLOW_NO_STOP(agent, Hash40::new("kamui_ryusensya_wind"), Hash40::new("arml"), 0, 0, 0, 0, 0, 0, 1, false);
                            }
                        }
                    }
                }
            }
        }
    }
}
