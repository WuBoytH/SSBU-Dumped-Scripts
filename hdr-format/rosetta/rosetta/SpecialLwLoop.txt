unsafe extern "C" fn game_speciallwloop(agent: &mut L2CAgentBase) {
    let boma = agent.boma();
    let lua_state = agent.lua_state_agent;
    if is_excute(agent) {
        ItemModule::pickup_item(boma, *ITEM_SIZE_LIGHT, *FIGHTER_HAVE_ITEM_WORK_TEMPORARY, *ITEM_TRAIT_FLAG_QUICK, 0, 0);
        ItemModule::use_item(boma, *FIGHTER_HAVE_ITEM_WORK_TEMPORARY, false);
        ItemModule::pickup_item(boma, *ITEM_SIZE_LIGHT, 0, -1, 0, 0);
    }
    wait(lua_state, 1.0);
}
