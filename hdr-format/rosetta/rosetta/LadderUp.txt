unsafe extern "C" fn sound_ladderup(agent: &mut L2CAgentBase) {
    let boma = agent.boma();
    let lua_state = agent.lua_state_agent;
    loop {
        if is_excute(agent) {
            PLAY_SE(agent, Hash40::new("se_rosetta_step_right_m"));
        }
        wait(lua_state, 5.0);
        if is_excute(agent) {
            PLAY_SE(agent, Hash40::new("se_rosetta_step_left_m"));
        }
        wait(lua_state, 5.0);
    }
}

unsafe extern "C" fn expression_ladderup(agent: &mut L2CAgentBase) {
    let boma = agent.boma();
    let lua_state = agent.lua_state_agent;
    loop {
        frame(lua_state, 1.0);
        if is_excute(agent) {
            ControlModule::set_rumble(boma, Hash40::new("rbkind_walk"), 0, false, 0x50000000);
        }
        frame(lua_state, 6.0);
        if is_excute(agent) {
            ControlModule::set_rumble(boma, Hash40::new("rbkind_walk"), 0, false, 0x50000000);
        }
        wait_loop_sync_mot(lua_state);
    }
}
