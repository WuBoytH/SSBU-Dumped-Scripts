    if is_excute(agent) {
        CHECK_VALID_FINAL_START_CAMERA(agent, 0, 7, 20, 0, 0, 0);
        SLOW_OPPONENT(agent, 15.0, 70.0);
    }
    frame(lua_state, 1.0);
    FT_MOTION_RATE(agent, 0.5);
    if !WorkModule::is_flag(boma, *FIGHTER_INSTANCE_WORK_ID_FLAG_DISABLE_FINAL_START_CAMERA) {
        if is_excute(agent) {
            FT_SET_FINAL_FEAR_FACE(agent, 55);
        }
        if get_value_float(lua_state, *SO_VAR_FLOAT_LR) < 0.0 {
            if is_excute(agent) {
                REQ_FINAL_START_CAMERA(agent, Hash40::new("d04final02.nuanmb"), true);
            }
        } else {
            if is_excute(agent) {
                REQ_FINAL_START_CAMERA(agent, Hash40::new("d04final.nuanmb"), true);
            }
        }
    } else {
        if is_excute(agent) {
            let scale = PostureModule::scale(boma);

        }
    }
    if is_excute(agent) {
        FT_START_CUTIN(agent);
    }
    frame(lua_state, 50.0);
    if is_excute(agent) {
        CAM_ZOOM_OUT(agent);
    }
}
