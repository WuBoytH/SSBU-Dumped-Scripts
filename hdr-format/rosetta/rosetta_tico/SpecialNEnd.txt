unsafe extern "C" fn game_specialnend(agent: &mut L2CAgentBase) {
    let boma = agent.boma();
    let lua_state = agent.lua_state_agent;
    if is_excute(agent) {
        KineticModule::clear_speed_all(boma);
    }
    frame(lua_state, 1.0);
    FT_MOTION_RATE(agent, 0.6);
    frame(lua_state, 11.0);
    FT_MOTION_RATE(agent, 1);
}
