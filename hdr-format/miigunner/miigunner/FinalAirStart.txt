unsafe extern "C" fn game_finalairstart(agent: &mut L2CAgentBase) {
    let boma = agent.boma();
    let lua_state = agent.lua_state_agent;
    if is_excute(agent) {
        CHECK_VALID_FINAL_START_CAMERA(agent, 0, 7, 20, 43, 0, 0);
        SLOW_OPPONENT(agent, 8.0, 70.0);
        ArticleModule::generate_article(boma, *FIGHTER_MIIGUNNER_GENERATE_ARTICLE_LASER, false, -1);
        ArticleModule::generate_article(boma, *FIGHTER_MIIGUNNER_GENERATE_ARTICLE_FULLTHROTTLE, false, -1);
    }
    if !WorkModule::is_flag(boma, *FIGHTER_INSTANCE_WORK_ID_FLAG_DISABLE_FINAL_START_CAMERA) {
        frame(lua_state, 10.0);
        if is_excute(agent) {
            FT_SET_FINAL_FEAR_FACE(agent, 60);
        }
        if get_value_float(lua_state, *SO_VAR_FLOAT_LR) < 0.0 {
            if is_excute(agent) {
                REQ_FINAL_START_CAMERA(agent, Hash40::new("d04finalstart02.nuanmb"), true, false) /* TODO: macro expects 2 args */;
            }
        } else {
            if is_excute(agent) {
                REQ_FINAL_START_CAMERA(agent, Hash40::new("d04finalstart.nuanmb"), true, false) /* TODO: macro expects 2 args */;
            }
        }
    } else {
        if is_excute(agent) {
            let scale = PostureModule::scale(boma);
            CAM_ZOOM_IN(agent, 3, 0, scale * 2.2, 0, 0);
        }
    }
    if is_excute(agent) {
        FT_START_CUTIN(agent);
    }
    frame(lua_state, 30.0);
    if is_excute(agent) {
        CAM_ZOOM_OUT(agent);
    }
}
