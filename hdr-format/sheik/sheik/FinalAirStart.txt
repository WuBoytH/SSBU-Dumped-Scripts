unsafe extern "C" fn game_finalairstart(agent: &mut L2CAgentBase) {
    let boma = agent.boma();
    let lua_state = agent.lua_state_agent;
    frame(lua_state, 1.0);
    FT_MOTION_RATE(agent, 1);
    if is_excute(agent) {
        CHECK_VALID_FINAL_START_CAMERA(agent, 0, 7, 20, 0, 0, 0);
        SLOW_OPPONENT(agent, 10.0, 64.0);
    }
    if !WorkModule::is_flag(boma, *FIGHTER_INSTANCE_WORK_ID_FLAG_DISABLE_FINAL_START_CAMERA) {
        frame(lua_state, 10.0);
        if is_excute(agent) {
            FT_SET_FINAL_FEAR_FACE(agent, 60);
            REQ_FINAL_START_CAMERA(agent, Hash40::new("d04finalstart.nuanmb"), true);
            FT_START_CUTIN(agent);
        }
    } else {
        if is_excute(agent) {
            let scale = PostureModule::scale(boma);
            CAM_ZOOM_IN(agent, 10, 0, scale * 1.9, 0, 0);
            camera!(agent, *MA_MSC_CMD_CAMERA_CAM_OFFSET, 0, -5);
            FT_START_CUTIN(agent);
        }
    }
    frame(lua_state, 35.0);
    FT_MOTION_RATE(agent, 1);
    if is_excute(agent) {
        CAM_ZOOM_OUT(agent);
    }
}
