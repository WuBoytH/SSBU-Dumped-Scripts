unsafe extern "C" fn game_finalstart(agent: &mut L2CAgentBase) {
    let boma = agent.boma();
    let lua_state = agent.lua_state_agent;
    if is_excute(agent) {
        WHOLE_HIT(agent, *HIT_STATUS_XLU);
        SLOW_OPPONENT(agent, 6.0, 55.0);
        CHECK_VALID_FINAL_START_CAMERA(agent, 0, 7, 20, 0, 0, 0);
    }
    if !WorkModule::is_flag(boma, *FIGHTER_INSTANCE_WORK_ID_FLAG_DISABLE_FINAL_START_CAMERA) {
        frame(lua_state, 10.0);
        if is_excute(agent) {
            FT_SET_FINAL_FEAR_FACE(agent, 60);
            REQ_FINAL_START_CAMERA(agent, Hash40::new("d04finalstart.nuanmb"), false);
            FT_START_CUTIN(agent);
        }
        frame(lua_state, 35.0);
        if is_excute(agent) {
            SlowModule::set_whole(boma, 3, 0);
        }
    } else {
        if is_excute(agent) {
            let scale = PostureModule::scale(boma);
            CAM_ZOOM_IN(agent, 5, 0, scale * 2.0, -0.1, 0);
            FT_START_CUTIN(agent);
            camera!(agent, *MA_MSC_CMD_CAMERA_CAM_RECT, 0, 4, 6, 0);
            camera!(agent, *MA_MSC_CMD_CAMERA_CAM_OFFSET, 0, 6);
        }
        frame(lua_state, 30.0);
        if is_excute(agent) {
            CAM_ZOOM_OUT(agent);
        }
    }
    frame(lua_state, 41.0);
    if !WorkModule::is_flag(boma, *FIGHTER_INSTANCE_WORK_ID_FLAG_DISABLE_FINAL_START_CAMERA) {
        if is_excute(agent) {
            SlowModule::clear_whole(boma);
        }
    }
    if is_excute(agent) {
        camera!(agent, *MA_MSC_CMD_CAMERA_CAM_OFFSET, 0, 0);
    }
}
