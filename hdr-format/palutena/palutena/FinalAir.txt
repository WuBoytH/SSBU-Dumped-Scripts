unsafe extern "C" fn game_finalair(agent: &mut L2CAgentBase) {
    let boma = agent.boma();
    let lua_state = agent.lua_state_agent;
    FT_MOTION_RATE(agent, 0.8);
    frame(lua_state, 1.0);
    if is_excute(agent) {
        CHECK_VALID_FINAL_START_CAMERA(agent, 0, 7, 20, 13, 0, 0);
        SLOW_OPPONENT(agent, 6.0, 85.0);
    }
    if !WorkModule::is_flag(boma, *FIGHTER_INSTANCE_WORK_ID_FLAG_DISABLE_FINAL_START_CAMERA) {
        frame(lua_state, 10.0);
        if is_excute(agent) {
            FT_SET_FINAL_FEAR_FACE(agent, 60);
            REQ_FINAL_START_CAMERA(agent, Hash40::new("d04final.nuanmb"), false, false) /* TODO: macro expects 2 args */;
            FT_START_CUTIN(agent);
        }
    } else {
        if is_excute(agent) {
            let scale = PostureModule::scale(boma);
            CAM_ZOOM_IN(agent, 10, 0, scale * 2.0, 0, 0);
            camera!(agent, *MA_MSC_CMD_CAMERA_CAM_OFFSET, 0, 6);
            FT_START_CUTIN(agent);
        }
    }
    frame(lua_state, 40.0);
    if is_excute(agent) {
        CAM_ZOOM_OUT(agent);
    }
    frame(lua_state, 70.0);
    if is_excute(agent) {
        camera!(agent, *MA_MSC_CMD_CAMERA_CAM_RECT, 25, -10, 10, -15);
    }
    frame(lua_state, 90.0);
    FT_MOTION_RATE(agent, 0.4);
}
