    let lua_state = agent.lua_state_agent;
    frame(lua_state, 2.0);
    if is_excute(agent) {
        SEARCH(agent, 0, 0, Hash40::new("rot"), 5.0, 0.0, 0.0, 0.0, None, None, None, *COLLISION_KIND_MASK_HIT, *HIT_STATUS_MASK_NORMAL, 1, *COLLISION_SITUATION_MASK_GA, *COLLISION_CATEGORY_MASK_FIGHTER, *COLLISION_PART_MASK_ALL, true);
    }
    if !WorkModule::is_flag(boma, *WEAPON_SNAKE_C4_INSTANCE_WORK_ID_FLAG_SQUAT) {
        frame(lua_state, 8.0);
        if is_excute(agent) {
            WorkModule::on_flag(boma, *WEAPON_SNAKE_C4_STATUS_ESTABLISH_WALL_WORK_FLAG_START_CHECK_STICK);
        }
    } else {
        frame(lua_state, 5.0);
        if is_excute(agent) {
            WorkModule::on_flag(boma, *WEAPON_SNAKE_C4_STATUS_ESTABLISH_WALL_WORK_FLAG_START_CHECK_STICK);
        }
    }
}
