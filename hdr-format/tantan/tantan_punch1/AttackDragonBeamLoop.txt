    if IS_EXIST_ARTICLE(agent, *WEAPON_TANTAN_PUNCH1_GENERATE_ARTICLE_BEAM) {

            ArticleModule::remove_exist(boma, *WEAPON_TANTAN_PUNCH1_GENERATE_ARTICLE_BEAM, ArticleOperationTarget(0));
        }

    }
}
