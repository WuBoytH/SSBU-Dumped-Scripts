unsafe extern "C" fn game_specialnstart(agent: &mut L2CAgentBase) {
    let boma = agent.boma();
    let lua_state = agent.lua_state_agent;
    if is_excute(agent) {
        ArticleModule::generate_article(boma, *FIGHTER_TANTAN_GENERATE_ARTICLE_SPIRALSIMPLE, false, -1);
        ArticleModule::set_visibility_whole(boma, *FIGHTER_TANTAN_GENERATE_ARTICLE_SPIRALSIMPLE, false, ArticleOperationTarget(0));
    }
    frame(lua_state, 3.0);
    if is_excute(agent) {
        ArticleModule::set_visibility_whole(boma, *FIGHTER_TANTAN_GENERATE_ARTICLE_SPIRALSIMPLE, true, ArticleOperationTarget(0));
    }
}
