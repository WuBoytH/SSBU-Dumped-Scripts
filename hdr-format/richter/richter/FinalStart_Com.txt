unsafe extern "C" fn game_finalstart_com(agent: &mut L2CAgentBase) {
    let boma = agent.boma();
    let lua_state = agent.lua_state_agent;
    if is_excute(agent) {
        CHECK_VALID_FINAL_START_CAMERA(agent, 0, 7, 20, 0, 0, 0);
        SLOW_OPPONENT(agent, 4.0, 50.0);
    }
    if !WorkModule::is_flag(boma, *FIGHTER_INSTANCE_WORK_ID_FLAG_DISABLE_FINAL_START_CAMERA) {
        frame(lua_state, 5.0);
        if is_excute(agent) {
            SlowModule::set_whole(boma, 2, 0);
            FT_SET_FINAL_FEAR_FACE(agent, 40);
        }
        if get_value_float(lua_state, *SO_VAR_FLOAT_LR) < 0.0 {
            if is_excute(agent) {
                REQ_FINAL_START_CAMERA(agent, Hash40::new("d04finalstart02.nuanmb"), false, false) /* TODO: macro expects 2 args */;
            }
        } else {
            if is_excute(agent) {
                REQ_FINAL_START_CAMERA(agent, Hash40::new("d04finalstart.nuanmb"), false, false) /* TODO: macro expects 2 args */;
            }
        }
    } else {
        if is_excute(agent) {
            let scale = PostureModule::scale(boma);
            CAM_ZOOM_IN(agent, 5, 0, scale * 2.1, 0, 0);
        }
    }
    if is_excute(agent) {
        FT_START_CUTIN(agent);
    }
    frame(lua_state, 29.0);
    if is_excute(agent) {
        SlowModule::clear_whole(boma);
    }
    frame(lua_state, 30.0);
    if is_excute(agent) {
        ArticleModule::set_visibility_whole(boma, *FIGHTER_SIMON_GENERATE_ARTICLE_COFFIN, true, ArticleOperationTarget(0));
        CAM_ZOOM_OUT(agent);
    }
}
