unsafe extern "C" fn game_finalr(agent: &mut L2CAgentBase) {
    let boma = agent.boma();
    let lua_state = agent.lua_state_agent;
    if is_excute(agent) {
        CHECK_VALID_FINAL_START_CAMERA(agent, 0, 7, 20, 0, 0, 0);
        SLOW_OPPONENT(agent, 5.0, 20.0);
    }
    if !WorkModule::is_flag(boma, *FIGHTER_INSTANCE_WORK_ID_FLAG_DISABLE_FINAL_START_CAMERA) {
        frame(lua_state, 5.0);
        if is_excute(agent) {
            FT_SET_FINAL_FEAR_FACE(agent, 40);
            REQ_FINAL_START_CAMERA(agent, Hash40::new("d04finalr.nuanmb"), false, false) /* TODO: macro expects 2 args */;
            FT_START_CUTIN(agent);
        }
    } else {
        if is_excute(agent) {
            let scale = PostureModule::scale(boma);
            CAM_ZOOM_IN(agent, 3, 0, scale * 2.2, 0, 0);
            FT_START_CUTIN(agent);
        }
    }
}
