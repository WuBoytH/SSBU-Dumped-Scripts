unsafe extern "C" fn expression_jumpfrontmini(agent: &mut L2CAgentBase) {
    let boma = agent.boma();
    let lua_state = agent.lua_state_agent;
    if IS_FIGHTER_STATUS_KIND(agent, *FIGHTER_STATUS_KIND_GIMMICK_SPRING_JUMP) {
        if is_excute(agent) {
            ControlModule::set_rumble(boma, Hash40::new("rbkind_bounce"), 0, true, 0x50000000);
        }
    } else if IS_FIGHTER_STATUS_KIND(agent, *FIGHTER_STATUS_KIND_ICE_JUMP) {
        if is_excute(agent) {
            ControlModule::set_rumble(boma, Hash40::new("rbkind_awaken"), 0, false, 0x50000000);
        }
    } else if IS_FIGHTER_STATUS_KIND(agent, *FIGHTER_STATUS_KIND_ITEM_ROCKETBELT_HOP) {
        if is_excute(agent) {
            ControlModule::set_rumble(boma, Hash40::new("rbkind_grapple"), 0, false, 0x50000000);
        }
    } else {
        if is_excute(agent) {
            ControlModule::set_rumble(boma, Hash40::new("rbkind_jump"), 0, false, 0x50000000);
        }
    }
}
