unsafe extern "C" fn game_itemlightget(agent: &mut L2CAgentBase) {
    let boma = agent.boma();
    let lua_state = agent.lua_state_agent;
    frame(lua_state, 3.0);
    if is_excute(agent) {
        ItemModule::pickup_item(boma, *ITEM_SIZE_LIGHT, 0, -1, 0, 0);
    }
}

unsafe extern "C" fn expression_itemlightget(agent: &mut L2CAgentBase) {
    let boma = agent.boma();
    let lua_state = agent.lua_state_agent;
    if is_excute(agent) {
        slope!(agent, *MA_MSC_CMD_SLOPE_SLOPE, *SLOPE_STATUS_LR);
    }
    frame(lua_state, 4.0);
    if ItemModule::is_have_item(boma, 0) {
        if is_excute(agent) {
            FT_MOTION_INTP_WAIT_ITEM(agent, true, 0.1);
        }
        frame(lua_state, 9.0);
        if is_excute(agent) {
            FT_MOTION_INTP_WAIT_ITEM(agent, true, 0);
        }
    }
}
