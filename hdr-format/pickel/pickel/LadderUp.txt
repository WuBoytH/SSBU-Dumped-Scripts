    let boma = agent.boma();
    let lua_state = agent.lua_state_agent;
    loop {
        if is_excute(agent) {
            PLAY_SE(agent, Hash40::new("se_pickel_ladderstep"));
        }
        wait(lua_state, 16.0);
    }
}
