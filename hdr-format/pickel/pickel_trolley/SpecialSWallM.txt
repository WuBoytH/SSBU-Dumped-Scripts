    if is_excute(agent) {
        ControlModule::set_rumble(boma, Hash40::new("rbkind_nohitl"), 0, false, 0x50000000);

    }
}

    let lua_state = agent.lua_state_agent;
    if WorkModule::is_flag(boma, *WEAPON_PICKEL_TROLLEY_INSTANCE_WORK_ID_FLAG_IS_WALL_LEFT) {
        if get_value_float(lua_state, *SO_VAR_FLOAT_LR) < 0.0 {
            if is_excute(agent) {
                EFFECT(agent, Hash40::new("sys_crown"), Hash40::new("top"), 0, 3, 6.5, -90, 0, 0, 0.7, 0, 0, 0, 0, 0, 0, true);
            }
        } else {
            if is_excute(agent) {
                EFFECT(agent, Hash40::new("sys_crown"), Hash40::new("top"), 0, 3, -6.5, 90, 0, 0, 0.7, 0, 0, 0, 0, 0, 0, true);
            }
        }
    } else if get_value_float(lua_state, *SO_VAR_FLOAT_LR) < 0.0 {
        if is_excute(agent) {
            EFFECT(agent, Hash40::new("sys_crown"), Hash40::new("top"), 0, 3, -6.5, 90, 0, 0, 0.7, 0, 0, 0, 0, 0, 0, true);
        }
    } else {
        if is_excute(agent) {
            EFFECT(agent, Hash40::new("sys_crown"), Hash40::new("top"), 0, 3, 6.5, -90, 0, 0, 0.7, 0, 0, 0, 0, 0, 0, true);
        }
    }

        LAST_EFFECT_SET_RATE(agent, 1.3);
    }
}
