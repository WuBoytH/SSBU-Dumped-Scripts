unsafe extern "C" fn effect_specialhilanding(agent: &mut L2CAgentBase) {
    let boma = agent.boma();
    let lua_state = agent.lua_state_agent;
    if get_value_float(lua_state, *SO_VAR_FLOAT_LR) < 0.0 {
        if is_excute(agent) {
            EFFECT(agent, Hash40::new("inkling_superjump_drop"), Hash40::new("top"), 0, 0, 0, 0, 180, 0, 1.05, 0, 0, 0, 0, 0, 0, false);
        }
        WorkModule::get_float(boma, *WEAPON_INKLING_SQUID_INSTANCE_WORK_ID_FLOAT_R);
        WorkModule::get_float(boma, *WEAPON_INKLING_SQUID_INSTANCE_WORK_ID_FLOAT_G);
        WorkModule::get_float(boma, *WEAPON_INKLING_SQUID_INSTANCE_WORK_ID_FLOAT_B);
        if is_excute(agent) {
            LAST_PARTICLE_SET_COLOR(agent) /* TODO: macro expects 3 args */;
        }
    } else {
        if is_excute(agent) {
            EFFECT(agent, Hash40::new("inkling_superjump_drop"), Hash40::new("top"), 0, 0, 0, 0, 0, 0, 1.05, 0, 0, 0, 0, 0, 0, false);
        }
        WorkModule::get_float(boma, *WEAPON_INKLING_SQUID_INSTANCE_WORK_ID_FLOAT_R);
        WorkModule::get_float(boma, *WEAPON_INKLING_SQUID_INSTANCE_WORK_ID_FLOAT_G);
        WorkModule::get_float(boma, *WEAPON_INKLING_SQUID_INSTANCE_WORK_ID_FLOAT_B);
        if is_excute(agent) {
            LAST_PARTICLE_SET_COLOR(agent) /* TODO: macro expects 3 args */;
        }
    }
}
