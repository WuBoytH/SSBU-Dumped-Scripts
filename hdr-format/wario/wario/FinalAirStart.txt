unsafe extern "C" fn game_finalairstart(agent: &mut L2CAgentBase) {
    let boma = agent.boma();
    let lua_state = agent.lua_state_agent;
    FT_MOTION_RATE(agent, 1.1);
    if is_excute(agent) {
        ArticleModule::generate_article(boma, *FIGHTER_WARIO_GENERATE_ARTICLE_GARLIC, true, -1);
        CHECK_VALID_FINAL_START_CAMERA(agent, 0, 7, 20, 0, 0, 0);
        SLOW_OPPONENT(agent, 20.0, 55.0);
    }
    if !WorkModule::is_flag(boma, *FIGHTER_INSTANCE_WORK_ID_FLAG_DISABLE_FINAL_START_CAMERA) {
        frame(lua_state, 9.0);
        if is_excute(agent) {
            FT_SET_FINAL_FEAR_FACE(agent, 75);
            REQ_FINAL_START_CAMERA(agent, Hash40::new("d04finalstart.nuanmb"), false);
            FT_START_CUTIN(agent);
        }
    } else {
        if is_excute(agent) {
            let scale = PostureModule::scale(boma);
            CAM_ZOOM_IN(agent, 4, 0, scale * 2.1, 0, 0);
            FT_START_CUTIN(agent);
        }
    }
    frame(lua_state, 18.0);
    if is_excute(agent) {
        ArticleModule::remove_exist(boma, *FIGHTER_WARIO_GENERATE_ARTICLE_GARLIC, ArticleOperationTarget(0));
    }
    if WorkModule::is_flag(boma, *FIGHTER_INSTANCE_WORK_ID_FLAG_DISABLE_FINAL_START_CAMERA) {
        frame(lua_state, 25.0);
        if is_excute(agent) {
            CAM_ZOOM_OUT(agent);
        }
    }
}
