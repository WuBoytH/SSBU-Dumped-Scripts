unsafe extern "C" fn sound_speciallwturn(agent: &mut L2CAgentBase) {
    let boma = agent.boma();
    let lua_state = agent.lua_state_agent;
    frame(lua_state, 2.0);
    if is_excute(agent) {
        PLAY_STEP(agent, Hash40::new("se_dedede_step_right_m"));
    }
    wait(lua_state, 10.0);
    if is_excute(agent) {
        PLAY_STEP(agent, Hash40::new("se_dedede_step_left_m"));
    }
}
