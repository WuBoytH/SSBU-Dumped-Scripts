unsafe extern "C" fn game_finalstartl(agent: &mut L2CAgentBase) {
    let boma = agent.boma();
    let lua_state = agent.lua_state_agent;
    if is_excute(agent) {
        MotionModule::set_rate(boma, 1.5);
    }
    frame(lua_state, 1.0);
    if is_excute(agent) {
        FT_START_CUTIN(agent);
        ArticleModule::change_status(boma, *FIGHTER_PIKMIN_GENERATE_ARTICLE_DOLFIN, *WEAPON_PIKMIN_DOLFIN_STATUS_KIND_START, ArticleOperationTarget(0));
        SLOW_OPPONENT(agent, 5.0, 40.0);
        CHECK_VALID_FINAL_START_CAMERA(agent, 0, 7, 20, 0, 0, 0);
    }
    if !WorkModule::is_flag(boma, *FIGHTER_INSTANCE_WORK_ID_FLAG_DISABLE_FINAL_START_CAMERA) {
        if is_excute(agent) {
            FT_SET_FINAL_FEAR_FACE(agent, 60);
            REQ_FINAL_START_CAMERA(agent, Hash40::new("d04finalstartr.nuanmb"), false, false) /* TODO: macro expects 2 args */;
        }
    } else {
        if is_excute(agent) {
            let scale = PostureModule::scale(boma);
            CAM_ZOOM_IN(agent, 5, 0, scale * 2.2, 0, 0);
        }
    }
    frame(lua_state, 41.0);
    if is_excute(agent) {
        MotionModule::set_rate(boma, 1.0);
        CAM_ZOOM_OUT(agent);
    }
}
