unsafe extern "C" fn game_entry(agent: &mut L2CAgentBase) {
    let boma = agent.boma();
    let lua_state = agent.lua_state_agent;
    if is_excute(agent) {
        ArticleModule::generate_article(boma, *WEAPON_PTRAINER_PTRAINER_GENERATE_ARTICLE_MBALL, false, -1);
        ArticleModule::set_visibility_whole(boma, *WEAPON_PTRAINER_PTRAINER_GENERATE_ARTICLE_MBALL, false, ArticleOperationTarget(0));
    }
    frame(lua_state, 4.0);
    if is_excute(agent) {
        VisibilityModule::set_int64(boma, hash40("weapon"), hash40("none"));
        ArticleModule::set_visibility_whole(boma, *WEAPON_PTRAINER_PTRAINER_GENERATE_ARTICLE_MBALL, true, ArticleOperationTarget(0));
        ArticleModule::shoot(boma, *WEAPON_PTRAINER_PTRAINER_GENERATE_ARTICLE_MBALL, ArticleOperationTarget(*ARTICLE_OPE_TARGET_ALL), false);
    }
    frame(lua_state, 90.0);
    if is_excute(agent) {
        VisibilityModule::set_int64(boma, hash40("weapon"), hash40("m_ball_r"));
    }
}
