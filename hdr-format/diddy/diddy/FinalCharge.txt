unsafe extern "C" fn game_finalcharge(agent: &mut L2CAgentBase) {
    let boma = agent.boma();
    let lua_state = agent.lua_state_agent;
    if is_excute(agent) {
        let scale = PostureModule::scale(boma);
        CAM_ZOOM_IN(agent, 10, 0, scale * 2.1, 0, 0);
        SLOW_OPPONENT(agent, 30.0, 60.0);
    }
    frame(lua_state, 40.0);
    if is_excute(agent) {
        CAM_ZOOM_OUT(agent);
        WorkModule::on_flag(boma, *FIGHTER_DIDDY_STATUS_FINAL_FLAG_END_CHARGE_ZOOM);
    }
}
