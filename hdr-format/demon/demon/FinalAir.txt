unsafe extern "C" fn game_finalair(agent: &mut L2CAgentBase) {
    let boma = agent.boma();
    let lua_state = agent.lua_state_agent;
    if is_excute(agent) {
        WHOLE_HIT(agent, *HIT_STATUS_XLU);
    }
    frame(lua_state, 1.0);
    if WorkModule::is_flag(boma, *FIGHTER_DEMON_STATUS_FINAL_FLAG_DEMON_DOJO_ZOOM) {
        if get_value_float(lua_state, *SO_VAR_FLOAT_LR) < 0.0 {
            if is_excute(agent) {
                let scale = PostureModule::scale(boma);
                CAM_ZOOM_IN(agent, 1, 0, scale * 2.0, 0, -0.1);
                camera!(agent, *MA_MSC_CMD_CAMERA_CAM_OFFSET, -5, 5);
            }
        } else {
            if is_excute(agent) {
                let scale = PostureModule::scale(boma);
                CAM_ZOOM_IN(agent, 1, 0, scale * 2.0, 0, 0.1);
                camera!(agent, *MA_MSC_CMD_CAMERA_CAM_OFFSET, 5, 5);
            }
        }
    } else if get_value_float(lua_state, *SO_VAR_FLOAT_LR) < 0.0 {
        if is_excute(agent) {
            let scale = PostureModule::scale(boma);
            CAM_ZOOM_IN(agent, 1, 0, scale * 2.0, 0, -0.3);
            camera!(agent, *MA_MSC_CMD_CAMERA_CAM_OFFSET, -5, 5);
        }
    } else {
        if is_excute(agent) {
            let scale = PostureModule::scale(boma);
            CAM_ZOOM_IN(agent, 1, 0, scale * 2.0, 0, 0.3);
            camera!(agent, *MA_MSC_CMD_CAMERA_CAM_OFFSET, 5, 5);
        }
    }
    frame(lua_state, 3.0);
    if is_excute(agent) {
        WorkModule::on_flag(boma, *FIGHTER_DEMON_STATUS_FINAL_FLAG_END_FINAL);
    }
    frame(lua_state, 5.0);
    if is_excute(agent) {
        CAM_ZOOM_OUT(agent);
    }
    frame(lua_state, 15.0);
    if is_excute(agent) {
        WorkModule::on_flag(boma, *FIGHTER_STATUS_VISUAL_SCENE_FLAG_END_EXIT);
    }
}

    let boma = agent.boma();

    if is_excute(agent) {
        CANCEL_FILL_SCREEN(agent, 1, 0);
        CANCEL_FILL_SCREEN(agent, 0, 0);
    }
    if get_value_float(lua_state, *SO_VAR_FLOAT_LR) < 0.0 {
        if is_excute(agent) {
            EffectModule::req_screen(boma, Hash40::new("bg_demon_final_after_l"), false, false, false);
        }
    } else {
        if is_excute(agent) {
            EffectModule::req_screen(boma, Hash40::new("bg_demon_final_after_r"), false, false, false);
        }
    }
    if is_excute(agent) {
        EFFECT_OFF_KIND(agent, Hash40::new("demon_final_smoke"), true, true);
        EFFECT_OFF_KIND(agent, Hash40::new("demon_final_sparks"), true, true);
        EFFECT_FOLLOW(agent, Hash40::new("demon_final_wind"), Hash40::new("top"), 0, 0, 0, 0, 0, 0, 1, true);

    frame(lua_state, 4.0);

            EffectModule::req_screen(boma, Hash40::new("bg_demon_final_after_l"), false, false, false);

            EffectModule::req_screen(boma, Hash40::new("bg_demon_final_after_r"), false, false, false);
        }
    }
}
