unsafe extern "C" fn sound_swim(agent: &mut L2CAgentBase) {
    let boma = agent.boma();
    let lua_state = agent.lua_state_agent;
    loop {
        frame(lua_state, 20.0);
        if is_excute(agent) {
            PLAY_SE(agent, Hash40::new("se_common_swim_high_02"));
        }
        wait(lua_state, 20.0);
        wait_loop_sync_mot(lua_state);
    }
}
