unsafe extern "C" fn game_finalstart(agent: &mut L2CAgentBase) {
    let boma = agent.boma();
    let lua_state = agent.lua_state_agent;
    if is_excute(agent) {
        SLOW_OPPONENT(agent, 10.0, 60.0);
        FT_REMOVE_FINAL_AURA(agent, true);
        FT_SET_FINAL_FEAR_FACE(agent, 70);
        let scale = PostureModule::scale(boma);
        CAM_ZOOM_IN(agent, 1, 0, scale * 2.1, 0, 0);
    }
    frame(lua_state, 5.0);
    if is_excute(agent) {
        FT_START_CUTIN(agent);
    }
    frame(lua_state, 12.0);
    if is_excute(agent) {
        CAM_ZOOM_OUT(agent);
        let scale = PostureModule::scale(boma);
        CAM_ZOOM_IN(agent, 15, 0, scale * 3.5, 0, 0);
    }
    if get_value_float(lua_state, *SO_VAR_FLOAT_LR) < 0.0 {
        if is_excute(agent) {
            camera!(agent, *MA_MSC_CMD_CAMERA_CAM_OFFSET, -10, 5);
        }
    } else {
        if is_excute(agent) {
            camera!(agent, *MA_MSC_CMD_CAMERA_CAM_OFFSET, 10, 5);
        }
    }
    frame(lua_state, 60.0);
    if is_excute(agent) {
        CAM_ZOOM_OUT(agent);
    }
}
