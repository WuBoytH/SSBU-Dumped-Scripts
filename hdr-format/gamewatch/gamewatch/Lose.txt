unsafe extern "C" fn expression_lose(agent: &mut L2CAgentBase) {
    let boma = agent.boma();
    let lua_state = agent.lua_state_agent;
    if is_excute(agent) {
        VisibilityModule::set_mesh_visibility(boma, Hash40::new("head"), *MESH_VISIBILITY_NONE, false);
        VisibilityModule::set_mesh_visibility(boma, Hash40::new("lhand"), *MESH_VISIBILITY_NONE, false);
        VisibilityModule::set_mesh_visibility(boma, Hash40::new("rhand"), *MESH_VISIBILITY_NONE, false);
        VisibilityModule::set_mesh_visibility(boma, Hash40::new("hand"), *MESH_VISIBILITY_NONE, false);
    }
}
