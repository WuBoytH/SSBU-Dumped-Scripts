    let boma = agent.boma();
    let lua_state = agent.lua_state_agent;
    frame(lua_state, 1.0);
    if is_excute(agent) {
        MotionModule::set_frame_partial(boma, *WEAPON_PACMAN_BIGPACMAN_MOTION_PART_SET_KIND_MATERIAL, 0.0, true);
        MotionModule::set_rate_partial(boma, *WEAPON_PACMAN_BIGPACMAN_MOTION_PART_SET_KIND_MATERIAL, 10.0);
    }
    frame(lua_state, 4.0);
    if is_excute(agent) {
        MotionModule::set_rate_partial(boma, *WEAPON_PACMAN_BIGPACMAN_MOTION_PART_SET_KIND_MATERIAL, 0.0);
    }
}
