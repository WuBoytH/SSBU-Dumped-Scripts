unsafe extern "C" fn game_bound(agent: &mut L2CAgentBase) {
    let boma = agent.boma();
    let lua_state = agent.lua_state_agent;
    if is_excute(agent) {
        ATK_POWER(agent, 0, 15);
        ATK_POWER(agent, 1, 15);
    }
}

unsafe extern "C" fn effect_bound(agent: &mut L2CAgentBase) {
    let boma = agent.boma();
    let lua_state = agent.lua_state_agent;
    if is_excute(agent) {
        QUAKE(agent, *CAMERA_QUAKE_KIND_M);
    }
    if WorkModule::get_int64(boma, *WEAPON_MURABITO_TREE_STATUS_FALLEN_WORK_INT_HIT_NUM) == 1 {
        if is_excute(agent) {
            let smoke_rot_x = WorkModule::get_float(boma, *WEAPON_MURABITO_TREE_INSTANCE_WORK_ID_FLOAT_SMOKE_ROT_X);
            let smoke_rot_y = WorkModule::get_float(boma, *WEAPON_MURABITO_TREE_INSTANCE_WORK_ID_FLOAT_SMOKE_ROT_Y);
            let smoke_rot_z = WorkModule::get_float(boma, *WEAPON_MURABITO_TREE_INSTANCE_WORK_ID_FLOAT_SMOKE_ROT_Z);
            EFFECT_WORK_R(agent, Hash40::new("murabito_tree_smoke"), Hash40::new("tree"), 0, 10, 0, smoke_rot_x, smoke_rot_y, smoke_rot_z, 1.2, 0, 0, 0, 0, 0, 0, true);
            let smoke_rot_x = WorkModule::get_float(boma, *WEAPON_MURABITO_TREE_INSTANCE_WORK_ID_FLOAT_SMOKE_ROT_X);
            let smoke_rot_y = WorkModule::get_float(boma, *WEAPON_MURABITO_TREE_INSTANCE_WORK_ID_FLOAT_SMOKE_ROT_Y);
            let smoke_rot_z = WorkModule::get_float(boma, *WEAPON_MURABITO_TREE_INSTANCE_WORK_ID_FLOAT_SMOKE_ROT_Z);
            EFFECT_WORK_R(agent, Hash40::new("murabito_leaf1"), Hash40::new("tree"), 0, 10, 0, smoke_rot_x, smoke_rot_y, smoke_rot_z, 1, 0, 0, 0, 0, 0, 0, true);
        }
    }
}

unsafe extern "C" fn sound_bound(agent: &mut L2CAgentBase) {
    let boma = agent.boma();
    let lua_state = agent.lua_state_agent;
    frame(lua_state, 0.0);
    if WorkModule::get_int64(boma, *WEAPON_MURABITO_TREE_STATUS_FALLEN_WORK_INT_HIT_NUM) == 1 {
        if is_excute(agent) {
            PLAY_SE(agent, Hash40::new("se_murabito_special_l08"));
        }
    } else {
        if is_excute(agent) {
            PLAY_SE(agent, Hash40::new("se_murabito_special_l09"));
            SET_PLAY_INHIVIT(agent, Hash40::new("se_murabito_special_l09"), 60);
        }
    }
}
