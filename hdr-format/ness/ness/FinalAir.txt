unsafe extern "C" fn game_finalair(agent: &mut L2CAgentBase) {
    let boma = agent.boma();
    let lua_state = agent.lua_state_agent;
    if is_excute(agent) {
        ArticleModule::generate_article(boma, *FIGHTER_NESS_GENERATE_ARTICLE_PAULA, false, -1);
        ArticleModule::generate_article(boma, *FIGHTER_NESS_GENERATE_ARTICLE_POO, false, -1);
    }
    if lua_bind::BattleObjectWorld::scale_z(singletons::BattleObjectWorld()) <= 1.0 {
        article!(agent, *MA_MSC_CMD_ARTICLE_SET_OFFSET_ARTICLE, *FIGHTER_NESS_GENERATE_ARTICLE_PAULA, -10, 0, -10);
        article!(agent, *MA_MSC_CMD_ARTICLE_SET_OFFSET_ARTICLE, *FIGHTER_NESS_GENERATE_ARTICLE_POO, 10, 0, -10);
    } else {
        article!(agent, *MA_MSC_CMD_ARTICLE_SET_OFFSET_ARTICLE, *FIGHTER_NESS_GENERATE_ARTICLE_PAULA, -10, 0, -0.01);
        article!(agent, *MA_MSC_CMD_ARTICLE_SET_OFFSET_ARTICLE, *FIGHTER_NESS_GENERATE_ARTICLE_POO, 10, 0, -0.01);
    }
    CHECK_VALID_FINAL_START_CAMERA(agent, 0, 7, 20, 0, 0, 0);
    SLOW_OPPONENT(agent, 5.0, 30.0);
    if !WorkModule::is_flag(boma, *FIGHTER_INSTANCE_WORK_ID_FLAG_DISABLE_FINAL_START_CAMERA) {
        frame(lua_state, 10.0);
        if is_excute(agent) {
            FT_SET_FINAL_FEAR_FACE(agent, 60);
            REQ_FINAL_START_CAMERA(agent, Hash40::new("d04final.nuanmb"), false);
            FT_START_CUTIN(agent);
        }
    } else {
        if is_excute(agent) {
            let scale = PostureModule::scale(boma);
            CAM_ZOOM_IN(agent, 5, 0, scale * 2.2, 0, 0);
            FT_START_CUTIN(agent);
        }
    }
    frame(lua_state, 25.0);
    if is_excute(agent) {
        CAM_ZOOM_OUT(agent);
        camera!(agent, *MA_MSC_CMD_CAMERA_CAM_RECT, 50, -50, 50, -10);
    }
    frame(lua_state, 51.0);
    if is_excute(agent) {
        WorkModule::on_flag(boma, *FIGHTER_NESS_INSTANCE_WORK_ID_FLAG_FINAL_START);
    }
    frame(lua_state, 350.0);
    if is_excute(agent) {
        ArticleModule::remove_exist(boma, *FIGHTER_NESS_GENERATE_ARTICLE_PAULA, ArticleOperationTarget(0));
        ArticleModule::remove_exist(boma, *FIGHTER_NESS_GENERATE_ARTICLE_POO, ArticleOperationTarget(0));
    }
    frame(lua_state, 357.0);
    if is_excute(agent) {
        cancel!(agent, *MA_MSC_CMD_CANCEL_ENABLE_CANCEL);
    }
}
