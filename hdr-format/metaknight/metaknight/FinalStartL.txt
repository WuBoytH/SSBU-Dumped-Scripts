unsafe extern "C" fn game_finalstartl(agent: &mut L2CAgentBase) {
    let boma = agent.boma();
    let lua_state = agent.lua_state_agent;
    if is_excute(agent) {
        WHOLE_HIT(agent, *HIT_STATUS_XLU);
        CHECK_VALID_FINAL_START_CAMERA(agent, 0, 7, 20, 12, 0, 0);
        SLOW_OPPONENT(agent, 10.0, 65.0);
    }
    if !WorkModule::is_flag(boma, *FIGHTER_INSTANCE_WORK_ID_FLAG_DISABLE_FINAL_START_CAMERA) {
        frame(lua_state, 10.0);
        if is_excute(agent) {
            ArticleModule::set_visibility_whole(boma, *FIGHTER_METAKNIGHT_GENERATE_ARTICLE_FOURWINGS, true, ArticleOperationTarget(0));
            FT_SET_FINAL_FEAR_FACE(agent, 60);
        }
        if get_value_float(lua_state, *SO_VAR_FLOAT_LR) < 0.0 {
            if is_excute(agent) {
                REQ_FINAL_START_CAMERA(agent, Hash40::new("d04finalstartl.nuanmb"), true, false) /* TODO: macro expects 2 args */;
            }
        } else {
            if is_excute(agent) {
                REQ_FINAL_START_CAMERA(agent, Hash40::new("d04finalstartr.nuanmb"), true, false) /* TODO: macro expects 2 args */;
            }
        }
    } else {
        if is_excute(agent) {
            FT_REMOVE_FINAL_AURA(agent, false);
        }
        frame(lua_state, 10.0);
        if is_excute(agent) {
            ArticleModule::set_visibility_whole(boma, *FIGHTER_METAKNIGHT_GENERATE_ARTICLE_FOURWINGS, true, ArticleOperationTarget(0));
            let scale = PostureModule::scale(boma);
            CAM_ZOOM_IN(agent, 5, 0, scale * 2.0, 0, 0);
            FT_START_CUTIN(agent);
        }
        frame(lua_state, 40.0);
        if is_excute(agent) {
            CAM_ZOOM_OUT(agent);
        }
    }
    if is_excute(agent) {
        FT_START_CUTIN(agent);
    }
}
