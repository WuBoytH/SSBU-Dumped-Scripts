unsafe extern "C" fn game_finalstart(agent: &mut L2CAgentBase) {
    let boma = agent.boma();
    let lua_state = agent.lua_state_agent;
    frame(lua_state, 1.0);
    FT_MOTION_RATE(agent, 1);
    if is_excute(agent) {
        CHECK_VALID_FINAL_START_CAMERA(agent, 0, 7, 20, 0, 0, 0);
        SLOW_OPPONENT(agent, 4.0, 60.0);
    }
    if !WorkModule::is_flag(boma, *FIGHTER_INSTANCE_WORK_ID_FLAG_DISABLE_FINAL_START_CAMERA) {
        if is_excute(agent) {
            FT_SET_FINAL_FEAR_FACE(agent, 60);
        }
        if get_value_float(lua_state, *SO_VAR_FLOAT_LR) < 0.0 {
            if is_excute(agent) {
                let scale = PostureModule::scale(boma);
                CAM_ZOOM_IN(agent, 6, 0, scale * 2.0, -0.3, 0);
                camera!(agent, *MA_MSC_CMD_CAMERA_CAM_OFFSET, -2, 8);
            }
        } else {
            if is_excute(agent) {
                let scale = PostureModule::scale(boma);
                CAM_ZOOM_IN(agent, 6, 0, scale * 2.0, -0.3, 0);
                camera!(agent, *MA_MSC_CMD_CAMERA_CAM_OFFSET, 2, 8);
            }
        }
    } else {
        if is_excute(agent) {
            let scale = PostureModule::scale(boma);
            CAM_ZOOM_IN(agent, 6, 0, scale * 2.0, 0, 0);
            FT_START_CUTIN(agent);
        }
    }
    if is_excute(agent) {
        FT_START_CUTIN(agent);
    }
    frame(lua_state, 15.0);
    if is_excute(agent) {
        SlowModule::set_whole(boma, 4, 0);
    }
    frame(lua_state, 16.0);
    if is_excute(agent) {
        EFFECT_OFF(agent, *EFFECT_SUB_ATTRIBUTE_EMIT, false);
        ArticleModule::set_visibility_whole(boma, *FIGHTER_SONIC_GENERATE_ARTICLE_SUPERSONIC, true, ArticleOperationTarget(0));
        ArticleModule::change_status(boma, *FIGHTER_SONIC_GENERATE_ARTICLE_SUPERSONIC, *WEAPON_SONIC_SUPERSONIC_STATUS_KIND_START, ArticleOperationTarget(0));
        VisibilityModule::set_whole(boma, false);
        SA_SET(agent, *SITUATION_KIND_AIR);
        CORRECT(agent, *GROUND_CORRECT_KIND_NONE);
    }
    frame(lua_state, 17.0);
    if is_excute(agent) {
        CAM_ZOOM_OUT(agent);
        let scale = PostureModule::scale(boma);
        CAM_ZOOM_IN(agent, 40, 0, scale * 2.0, 0.3, 0);
    }
    frame(lua_state, 20.0);
    if is_excute(agent) {
        SlowModule::clear_whole(boma);
    }
    frame(lua_state, 33.0);
    if is_excute(agent) {
        CAM_ZOOM_OUT(agent);
        let scale = PostureModule::scale(boma);
        CAM_ZOOM_IN(agent, 1, 0, scale * 2.0, 0, 0);
    }
    frame(lua_state, 35.0);
    if is_excute(agent) {
        WorkModule::on_flag(boma, *FIGHTER_SONIC_STATUS_FINAL_FLAG_FINAL_CAMERA);
    }
}
