unsafe extern "C" fn sound_swim(agent: &mut L2CAgentBase) {
    let boma = agent.boma();
    let lua_state = agent.lua_state_agent;
    loop {
        if is_excute(agent) {
            PLAY_SE(agent, Hash40::new("se_common_swim_low_01"));
        }
        wait(lua_state, 26.0);
        if is_excute(agent) {
            PLAY_SE(agent, Hash40::new("se_common_swim_middle_01"));
        }
        wait(lua_state, 26.0);
        PLAY_SE(agent, Hash40::new("se_common_swim_low_02"));
        wait_loop_sync_mot(lua_state);
    }
}
