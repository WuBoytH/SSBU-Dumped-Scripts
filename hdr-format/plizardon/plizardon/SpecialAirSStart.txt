    if is_excute(agent) {
        EFFECT_FOLLOW(agent, Hash40::new("plizardon_flare_blitz_hold"), Hash40::new("rot"), 0, 1, 0, 0, 0, 0, 1, true);
    }

        if is_excute(agent) {
            FLASH(agent, 1, 0.5, 0.5, 0.6);
        }
        wait(lua_state, 1.0);
        if is_excute(agent) {
            FLASH_FRM(agent, 2, 1, 0.3, 0.1, 0.7);
        }
        wait(lua_state, 2.0);

            COL_NORMAL(agent);
        }
        wait(lua_state, 1.0);
        if is_excute(agent) {
            BURN_COLOR(agent, 2, 0.1, 0, 0.4);
        }
        wait(lua_state, 1.0);
        if is_excute(agent) {
            BURN_COLOR_FRAME(agent, 2, 1, 0.2, 0.1, 0);
        }
        wait(lua_state, 1.0);

    }
}
