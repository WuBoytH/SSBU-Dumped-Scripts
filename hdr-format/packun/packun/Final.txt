unsafe extern "C" fn game_final(agent: &mut L2CAgentBase) {
    let boma = agent.boma();
    let lua_state = agent.lua_state_agent;
    if is_excute(agent) {
        SLOW_OPPONENT(agent, 2.0, 60.0);
        CHECK_VALID_FINAL_START_CAMERA(agent, 0, 7, 20, 0, 0, 0);
    }
    if !WorkModule::is_flag(boma, *FIGHTER_INSTANCE_WORK_ID_FLAG_DISABLE_FINAL_START_CAMERA) {
        frame(lua_state, 10.0);
        if is_excute(agent) {
            FT_SET_FINAL_FEAR_FACE(agent, 60);
            REQ_FINAL_START_CAMERA(agent, Hash40::new("d04final.nuanmb"), false, false) /* TODO: macro expects 2 args */;
            FT_START_CUTIN(agent);
        }
    } else {
        if is_excute(agent) {
            let scale = PostureModule::scale(boma);
            CAM_ZOOM_IN(agent, 3, 0, scale * 2.05, 0, 0);
            FT_START_CUTIN(agent);
        }
        if get_value_float(lua_state, *SO_VAR_FLOAT_LR) < 0.0 {
            if is_excute(agent) {
                camera!(agent, *MA_MSC_CMD_CAMERA_CAM_OFFSET, 0, 0);
            }
        } else {
            if is_excute(agent) {
                camera!(agent, *MA_MSC_CMD_CAMERA_CAM_OFFSET, 0, 0);
            }
        }
    }
    frame(lua_state, 25.0);
    if WorkModule::is_flag(boma, *FIGHTER_INSTANCE_WORK_ID_FLAG_DISABLE_FINAL_START_CAMERA) {
        if is_excute(agent) {
            CAM_ZOOM_OUT(agent);
        }
    }
    frame(lua_state, 30.0);
    if is_excute(agent) {
        CAM_ZOOM_OUT(agent);
    }
    if !IS_EXIST_ARTICLE(agent, *FIGHTER_PACKUN_GENERATE_ARTICLE_BOSSPACKUN) {
        if is_excute(agent) {
            ArticleModule::generate_article(boma, *FIGHTER_PACKUN_GENERATE_ARTICLE_BOSSPACKUN, false, -1);
        }
    }
}
