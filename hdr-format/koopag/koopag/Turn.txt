unsafe extern "C" fn game_turn(agent: &mut L2CAgentBase) {
    let boma = agent.boma();
    let lua_state = agent.lua_state_agent;
    frame(lua_state, 1.0);
    FT_MOTION_RATE(agent, 1.5);
    frame(lua_state, 7.0);
    FT_MOTION_RATE(agent, 2);
    frame(lua_state, 14.0);
    FT_MOTION_RATE(agent, 2.5);
}

unsafe extern "C" fn sound_turn(agent: &mut L2CAgentBase) {
    let boma = agent.boma();
    let lua_state = agent.lua_state_agent;
    frame(lua_state, 10.0);
    if is_excute(agent) {
        PLAY_STEP_FLIPPABLE(agent, Hash40::new("se_koopag_step_left_s"), Hash40::new("se_koopag_step_right_s"));
    }
}

unsafe extern "C" fn expression_turn(agent: &mut L2CAgentBase) {
    let boma = agent.boma();
    let lua_state = agent.lua_state_agent;
    if is_excute(agent) {
        slope!(agent, *MA_MSC_CMD_SLOPE_SLOPE, *SLOPE_STATUS_LR);
    }
    frame(lua_state, 12.0);
    if is_excute(agent) {
        QUAKE(agent, *CAMERA_QUAKE_KIND_S);
    }
}
