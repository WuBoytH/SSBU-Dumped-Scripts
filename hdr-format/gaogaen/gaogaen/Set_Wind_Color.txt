unsafe extern "C" fn effect_set_wind_color(agent: &mut L2CAgentBase) {
    let boma = agent.boma();
    let lua_state = agent.lua_state_agent;
    if WorkModule::get_int64(boma, *FIGHTER_INSTANCE_WORK_ID_INT_COLOR) == 0 {
        if is_excute(agent) {
            LAST_PARTICLE_SET_COLOR(agent, 1, 0.1, 0.1);
        }
    }
    if WorkModule::get_int64(boma, *FIGHTER_INSTANCE_WORK_ID_INT_COLOR) == 1 {
        if is_excute(agent) {
            LAST_PARTICLE_SET_COLOR(agent, 0.1, 1, 0.2);
        }
    }
    if WorkModule::get_int64(boma, *FIGHTER_INSTANCE_WORK_ID_INT_COLOR) == 2 {
        if is_excute(agent) {
            LAST_PARTICLE_SET_COLOR(agent, 0.8, 0.1, 0.1);
        }
    }
    if WorkModule::get_int64(boma, *FIGHTER_INSTANCE_WORK_ID_INT_COLOR) == 3 {
        if is_excute(agent) {
            LAST_PARTICLE_SET_COLOR(agent, 0.2, 0.4, 1);
        }
    }
    if WorkModule::get_int64(boma, *FIGHTER_INSTANCE_WORK_ID_INT_COLOR) == 4 {
        if is_excute(agent) {
            LAST_PARTICLE_SET_COLOR(agent, 1.2, 0.7, 0.1);
        }
    }
    if WorkModule::get_int64(boma, *FIGHTER_INSTANCE_WORK_ID_INT_COLOR) == 5 {
        if is_excute(agent) {
            LAST_PARTICLE_SET_COLOR(agent, 0.7, 0.1, 0.5);
        }
    }
    if WorkModule::get_int64(boma, *FIGHTER_INSTANCE_WORK_ID_INT_COLOR) == 6 {
        if is_excute(agent) {
            LAST_PARTICLE_SET_COLOR(agent, 0.4, 0.4, 0.4);
        }
    }
    if WorkModule::get_int64(boma, *FIGHTER_INSTANCE_WORK_ID_INT_COLOR) == 7 {
        if is_excute(agent) {
            LAST_PARTICLE_SET_COLOR(agent, 1.4, 0.3, 0.3);
        }
    }
}
