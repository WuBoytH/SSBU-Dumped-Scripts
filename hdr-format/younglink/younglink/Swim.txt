unsafe extern "C" fn sound_swim(agent: &mut L2CAgentBase) {
    let boma = agent.boma();
    let lua_state = agent.lua_state_agent;
    loop {
        frame(lua_state, 10.0);
        if is_excute(agent) {
            PLAY_SE(agent, Hash40::new("se_common_swim_middle_01"));
        }
        wait(lua_state, 35.0);
        if is_excute(agent) {
            PLAY_SE(agent, Hash40::new("se_common_swim_middle_02"));
        }
        wait_loop_sync_mot(lua_state);
    }
}

unsafe extern "C" fn expression_swim(agent: &mut L2CAgentBase) {
    let boma = agent.boma();
    let lua_state = agent.lua_state_agent;
    loop {
        if is_excute(agent) {
            VisibilityModule::set_int64(boma, hash40("shield"), hash40("shield_back"));
            VisibilityModule::set_int64(boma, hash40("sword"), hash40("sword_back"));
        }
        frame(lua_state, 12.0);
        if is_excute(agent) {
            ControlModule::set_rumble(boma, Hash40::new("rbkind_swim"), 0, false, 0x50000000);
        }
        frame(lua_state, 47.0);
        if is_excute(agent) {
            ControlModule::set_rumble(boma, Hash40::new("rbkind_swim"), 0, false, 0x50000000);
        }
        wait_loop_sync_mot(lua_state);
    }
}
